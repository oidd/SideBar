enum Clock { static var time = 0.0 }
struct Date: Comparable {
    let value: Double
    init() { value = Clock.time }
    init(_ value: Double) { self.value = value }
    static let distantPast = Date(-1e10)
    func timeIntervalSince(_ other: Date) -> Double { value - other.value }
    static func < (lhs: Date, rhs: Date) -> Bool { lhs.value < rhs.value }
}
struct Deadline {
    let value: Double
    static func now() -> Deadline { Deadline(value: Clock.time) }
    static func + (lhs: Deadline, rhs: Double) -> Deadline { Deadline(value: lhs.value + rhs) }
}
final class DispatchQueue {
    static let main = DispatchQueue()
    static var blocks: [(Double, () -> Void)] = []
    func asyncAfter(deadline: Deadline, execute: @escaping () -> Void) { Self.blocks.append((deadline.value, execute)) }
    static func drainDue() {
        let due = blocks.filter { $0.0 <= Clock.time }
        blocks.removeAll { $0.0 <= Clock.time }
        due.forEach { $0.1() }
    }
}
final class Timer {
    static var all: [Timer] = []
    var isValid = true
    let interval: Double
    var due: Double
    let action: (Timer) -> Void
    init(_ interval: Double, _ action: @escaping (Timer) -> Void) {
        self.interval = interval; due = Clock.time + interval; self.action = action
    }
    static func scheduledTimer(withTimeInterval interval: Double, repeats: Bool, block: @escaping (Timer) -> Void) -> Timer {
        precondition(repeats)
        let timer = Timer(interval, block); all.append(timer); return timer
    }
    func invalidate() { isValid = false }
    func tick() {
        guard isValid, due <= Clock.time + 1e-9 else { return }
        due = Clock.time + interval; action(self)
    }
}
func advance(_ ticks: Int) {
    for _ in 0..<ticks {
        Clock.time += 0.1
        DispatchQueue.drainDue()
        Timer.all.forEach { $0.tick() }
        DispatchQueue.drainDue()
    }
}
